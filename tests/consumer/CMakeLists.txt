cmake_minimum_required(VERSION 3.31)
project(ncei_consumer LANGUAGES CXX)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG -DNCEI_PARENT_FLAGS")
if(NCEI_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(ncei_cpp SOURCE_DIR "${NCEI_SOURCE_DIR}")
    FetchContent_MakeAvailable(ncei_cpp)
    if(NCEI_BUILD_TESTS OR NCEI_BUILD_EXAMPLES OR NCEI_ENABLE_LTO)
        message(FATAL_ERROR "SDK subproject defaults must be off")
    endif()
    get_directory_property(sdk_release_flags DIRECTORY "${NCEI_SOURCE_DIR}"
        DEFINITION CMAKE_CXX_FLAGS_RELEASE)
    if(NOT sdk_release_flags STREQUAL "-O2 -DNDEBUG -DNCEI_PARENT_FLAGS" OR
       NOT CMAKE_CXX_FLAGS_RELEASE STREQUAL "-O2 -DNDEBUG -DNCEI_PARENT_FLAGS")
        message(FATAL_ERROR "SDK overwrote the parent's release flags")
    endif()
else()
    find_package(ncei ${NCEI_EXPECTED_VERSION} EXACT CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE ncei::ncei)

if(README_EXAMPLES_DIR)
    file(GLOB examples "${README_EXAMPLES_DIR}/*.cpp")
    foreach(example IN LISTS examples)
        get_filename_component(name "${example}" NAME_WE)
        add_executable("${name}" "${example}")
        target_link_libraries("${name}" PRIVATE ncei::ncei)
    endforeach()
endif()
